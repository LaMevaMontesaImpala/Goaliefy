import { GoogleGenAI } from '@google/genai'; import { GoogleAIFileManager } from '@google/genai/files'; const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY }); const fileManager = new GoogleAIFileManager({ apiKey: process.env.GEMINI_API_KEY }); export async function handleVideoAnalysis(filePath, mimeType) { // 1. Cargar archivo a la Files API de Gemini const uploadResult = await fileManager.uploadFile(filePath, { mimeType: mimeType, displayName: "Goalkeeper Action Clip", }); // 2. Definir el System Prompt de Entrenador de Élite const systemInstruction = ` Eres un entrenador de porteros de hockey sobre hierba de nivel internacional. Analiza detenidamente el vídeo adjunto de la acción defensiva del portero. Debes responder ÚNICAMENTE con un JSON con la siguiente estructura: { "actionType": "Nombre exacto de la jugada", "decisionScore": 8, "decisionAnalysis": "Evaluación táctica del momento de decisión y posicionamiento.", "executionScore": 7, "executionAnalysis": "Análisis biomecánico: centro de gravedad, colocación del stick y uso de guardas/pies.", "strengths": ["Punto fuerte 1", "Punto fuerte 2"], "coachingTips": ["Consejo de mejora 1", "Ejercicio recomendado para corregir el error"] }`; // 3. Ejecutar análisis multimodal con Gemini 2.5 Flash const response = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: [ uploadResult.file, { text: "Realiza el análisis biomecánico y táctico completo de esta acción defensiva del portero." } ], config: { systemInstruction: systemInstruction, responseMimeType: "application/json", temperature: 0.2 } }); // 4. Limpieza del archivo en la nube await fileManager.deleteFile(uploadResult.file.name); return JSON.parse(response.text); }
