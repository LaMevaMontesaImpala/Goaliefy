// Archivo: api/analyze.js import { GoogleGenAI } from '@google/genai'; export default async function handler(req, res) { // Solo permitimos peticiones POST if (req.method !== 'POST') { return res.status(405).json({ error: 'Método no permitido' }); } try { const { imageBase64 } = req.body; if (!imageBase64) { return res.status(400).json({ error: 'No se ha recibido el fotograma de la jugada.' }); } // Inicializamos Gemini usando la variable de entorno de Vercel (totalmente oculta al usuario) const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY }); const systemInstruction = ` Eres un analista de rendimiento y entrenador de porteros de hockey sobre hierba de élite. Analiza este fotograma de una jugada defensiva. Clasifica la situación en UNA categoría táctica (ej: Penalty Corner, 1v1, Bola Aérea, etc). Evalúa del 1 al 10 la Toma de Decisiones y la Ejecución Técnica. Responde ÚNICAMENTE con un JSON válido con esta estructura: { "situation": "Nombre de la jugada", "decisionScore": 8, "decisionDesc": "Análisis táctico de la decisión.", "execScore": 7, "execDesc": "Análisis biomecánico (postura, stick, pies).", "strategy": "Estrategia de mejora." }`; const response = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: [ { text: systemInstruction }, { inlineData: { mimeType: "image/jpeg", data: imageBase64 } } ], config: { responseMimeType: "application/json", temperature: 0.2 } }); // Devolvemos el JSON limpio al frontend res.status(200).json(JSON.parse(response.text)); } catch (error) { console.error("Error en Backend:", error); res.status(500).json({ error: 'Error procesando la IA: ' + error.message }); } }
